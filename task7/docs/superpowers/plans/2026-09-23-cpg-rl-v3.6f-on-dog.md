```python
# task7/inference/export_policy_v3.py
"""brax PPO v3 權重（.pkl）→ 狗上 numpy 推論用的 .npz（M9 --v3 讀）。

用法：
    conda run -n rbtdog python task7/inference/export_policy_v3.py \
        --params task7/weights/cpg_rl_v3_6f_params.pkl
    → task7/weights/cpg_rl_v3_6f_np.npz

npz 內容：MLP（mean/std/W0..3/b0..3）＋ 產生器參數快照（ref_json／gen_json）＋ 相位表／幾何／限位／
原廠週期表（CYC_TURN_SYM 版）＋ 鏡像索引表。**狗端不自己算任何常數**，全部從這裡讀；
tests/test_export_policy_v3.py 釘住 npz == rl_env_v3 當下值。
"""
from __future__ import annotations

import argparse
import hashlib
import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import rl_env_v3 as v3  # noqa: E402

HIDDEN = ("hidden_0", "hidden_1", "hidden_2", "hidden_3")
# 模組名 → npz 鍵
TABLE_KEYS = {
    "PH_TURN_L": "PH_TURN_L", "PH_LAT_L": "PH_LAT_L", "PH_TROT": "PH_TROT",
    "W_LAT_L": "W_LAT_L", "W_TURN_L": "W_TURN_L", "W_ARC_L": "W_ARC_L", "DOM_TURN_L": "DOM_TURN_L", "LAT_WHEEL_L": "LAT_WHEEL_L",
    "MIRROR_LR": "MIRROR_LR", "SIDE_Y_j": "SIDE_Y", "FRONT_j": "FRONT", "FOOT_XY_BODY": "FOOT_XY_BODY",
    "STANCE_Q12_j": "STANCE_Q12", "STANCE_FEET_j": "STANCE_FEET", "KNEE_SIGN_j": "KNEE_SIGN4",
    "CYC_OFF_SYM": "CYC_OFF", "CYC_TAU_SYM": "CYC_TAU", "CYC_HZ_SYM": "CYC_HZ", "CYC_QMEAN": "CYC_QMEAN",
    "OBS_MIRROR_IDX": "OBS_MIRROR_IDX", "OBS_MIRROR_SGN": "OBS_MIRROR_SGN",
    "ACT_MIRROR_IDX": "ACT_MIRROR_IDX", "ACT_MIRROR_SGN": "ACT_MIRROR_SGN",
}
GAINS = {"kp3": [60.0, 120.0, 120.0], "kd3": [1.0, 1.0, 1.0], "wheel_kd": 0.1, "wheel_kd_safe": 0.5}


def _jsonable(v):
    if isinstance(v, (np.ndarray,)) or hasattr(v, "tolist"):
        return np.asarray(v).tolist()
    if isinstance(v, dict):
        return {k: _jsonable(x) for k, x in v.items()}
    if isinstance(v, (list, tuple)):
        return [_jsonable(x) for x in v]
    return v


def make_env():
    return v3.DualModeEnv(gains="factory", weights=v3.W36, ref=dict(cyc_amp_rand=False), push=False)


def export(pkl: str, out: str) -> dict:
    from brax.io import model
    params = model.load_params(pkl)
    norm, pol = params[0], params[1]["params"]
    mean, std = np.asarray(norm.mean, dtype=np.float32), np.asarray(norm.std, dtype=np.float32)
    assert mean.shape == (88,), mean.shape
    assert tuple(pol.keys()) == HIDDEN, tuple(pol.keys())
    Ws = [np.asarray(pol[h]["kernel"], dtype=np.float32) for h in HIDDEN]
    bs = [np.asarray(pol[h]["bias"], dtype=np.float32) for h in HIDDEN]
    assert Ws[0].shape == (88, 256) and Ws[3].shape == (128, 2 * v3.ACT_DIM), (Ws[0].shape, Ws[3].shape)
    env = make_env()
    K = dict(ACT_DIM=v3.ACT_DIM, QRES_MAX=v3.QRES_MAX, QRES_SLEW=v3.QRES_SLEW, OMEGA_NOM_SCALE=v3.OMEGA_NOM_SCALE,
             AMP_SCALE=v3.AMP_SCALE, LIFT_SCALE=v3.LIFT_SCALE, SWAY_MAX=v3.SWAY_MAX, SWAY_SLEW=v3.SWAY_SLEW,
             WHEEL_RES=v3.WHEEL_RES, STEP_GAIN_S=v3.STEP_GAIN_S, PH_SLEW=v3.PH_SLEW, FOOT_OFF_MAX=v3.FOOT_OFF_MAX,
             TAU_FF=v3.TAU_FF, RAMP_STEPS=v3.W36["RAMP_STEPS"], CTRL_DT=float(v3.CTRL_DT), CYC_N=int(v3.CYC_N),
             gains=GAINS, weights_preset="v36", cyc_turn_sym=True, obs_dim=88)
    z = dict(mean=mean, std=std, obs_dim=np.int64(88), act_dim=np.int64(v3.ACT_DIM),
             preset=np.str_("v36"), layout=np.str_("v3"),
             ref_json=np.str_(json.dumps(_jsonable(env.ref), sort_keys=True)),
             gen_json=np.str_(json.dumps(K, sort_keys=True)),
             src_sha256=np.str_(hashlib.sha256(Path(pkl).read_bytes()).hexdigest()),
             LO12=np.asarray(env._lo, dtype=np.float64), HI12=np.asarray(env._hi, dtype=np.float64))
    for i in range(4):
        z[f"W{i}"], z[f"b{i}"] = Ws[i], bs[i]
    for name, key in TABLE_KEYS.items():
        arr = np.asarray(getattr(v3, name))
        z[key] = arr.astype(np.int64) if arr.dtype.kind in "iu" else arr.astype(np.float64)
    np.savez(out, **z)
    return z


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--params", default=str(Path(__file__).resolve().parents[1] / "weights" / "cpg_rl_v3_6f_params.pkl"))
    ap.add_argument("--out", default="")
    a = ap.parse_args()
    out = a.out or str(Path(a.params).with_name(Path(a.params).stem.replace("_params", "_np") + ".npz"))
    z = export(a.params, out)
    print(f"✅ {out}  obs {int(z['obs_dim'])} act {int(z['act_dim'])} sha256 {str(z['src_sha256'])[:12]}…  鍵 {len(z)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# task7/realbot/gen_v3.py
"""v3 統一運動學產生器的 numpy 移植（狗上跑，2026-09-23）。

只 import numpy、json、標準庫與 realbot/kin.py。**本檔零常數**：所有數字由 npz 帶
（`load_params`），函式一對一對應 `inference/rl_env_v3.py` 的同名函式，
正確性由 `tests/test_gen_v3.py` 對 jax 版逐函式（2e-5）／逐步（q_des 1e-4 rad）釘住。
jax 端是 float32、這裡是 float64，所以容差不是 0。

`Gen.step(cmd, act, v_wheel)` 的順序逐行照 `DualModeEnv.step`（延遲是 env 的事，這裡吃「這一步要套用的動作」；
淡入 RAMP_STEPS 在這裡做，因為它是產生器狀態 i 的函數）。
"""
from __future__ import annotations

import json

import numpy as np

import kin

LEG_SHM = ("fr", "fl", "br", "bl")          # MJCF 序 (FR, FL, RR, RL) → kin.ik 的 shm 腿名
TWO_PI = 2.0 * np.pi
TABLE_KEYS = ("PH_TURN_L", "PH_LAT_L", "PH_TROT", "W_LAT_L", "W_TURN_L", "W_ARC_L", "DOM_TURN_L", "LAT_WHEEL_L",
              "MIRROR_LR", "SIDE_Y", "FRONT", "FOOT_XY_BODY", "STANCE_Q12", "STANCE_FEET", "KNEE_SIGN4", "LO12", "HI12",
              "CYC_OFF", "CYC_TAU", "CYC_HZ", "CYC_QMEAN")


class Params:
    """npz 快照：`ref`（env.ref dict）、`K`（模組常數 dict）、`T`（表，numpy）。"""

    def __init__(self, ref: dict, K: dict, T: dict):
        self.ref, self.K, self.T = ref, K, T
        assert ref["wheel_space"] == "tau", "v3.6f 只走力矩空間"
        assert not ref.get("cyc_lat_decouple", False), "v3.7 解耦不在本移植範圍"
        assert ref["cyc_recenter"], "cyc_recenter=False 的絕對模式沒移植"


def load_params(path: str) -> Params:
    with np.load(path, allow_pickle=False) as z:
        ref = json.loads(str(z["ref_json"]))
        K = json.loads(str(z["gen_json"]))
        T = {k: np.asarray(z[k], dtype=float) for k in TABLE_KEYS}
        T["MIRROR_LR"] = np.asarray(z["MIRROR_LR"]).astype(int)
    return Params(ref, K, T)


# ---------------------------------------------------------------- 純函式（對應 rl_env_v3）
def _mirror(v, right: bool, p: Params):
    return v[p.T["MIRROR_LR"]] if right else v


def activity(cmd, p: Params) -> dict:
    ar = p.ref["a_ref"]
    a_lat = float(np.clip(abs(cmd[1]) / ar["vy"], 0.0, 1.0))
    a_turn = float(np.clip(abs(cmd[2]) / ar["wz"], 0.0, 1.0))
    a_wheel = float(np.clip(abs(cmd[0]) / ar["vx"], 0.0, 1.0))
    return dict(lat=a_lat, turn=a_turn, wheel=a_wheel, s4=max(a_lat, a_turn * (1.0 - a_wheel)), arc=a_turn * a_wheel)


def lat_weight(A: dict) -> float:
    return A["lat"] / (A["lat"] + A["turn"] + 1e-6)


def leg_activity(cmd, A: dict, p: Params) -> np.ndarray:
    T = p.T
    w_lat = _mirror(T["W_LAT_L"], cmd[1] < 0, p)
    w_turn = _mirror(T["W_TURN_L"], cmd[2] < 0, p)
    w_arc = _mirror(T["W_ARC_L"], cmd[2] < 0, p)
    wl = lat_weight(A)
    return np.maximum(A["s4"] * (wl * w_lat + (1.0 - wl) * w_turn), A["arc"] * w_arc)


def step_gain(s, p: Params) -> np.ndarray:
    return np.clip(np.asarray(s, float) / p.K["STEP_GAIN_S"], 0.0, 1.0)


def phase_offsets(cmd, A: dict, p: Params) -> np.ndarray:
    T, ref = p.T, p.ref
    ph_t = T["PH_TROT"] if ref["phase_set_turn"] == "trot" else _mirror(T["PH_TURN_L"], cmd[2] < 0, p)
    ph_l = T["PH_TROT"] if ref["phase_set_lat"] == "trot" else _mirror(T["PH_LAT_L"], cmd[1] < 0, p)
    wl = lat_weight(A)
    z = wl * np.exp(2j * np.pi * ph_l) + (1.0 - wl) * np.exp(2j * np.pi * ph_t)
    return np.mod(np.angle(z) / TWO_PI, 1.0)


def slew_phase(ph, ph_tgt, p: Params) -> np.ndarray:
    d = np.mod(np.asarray(ph_tgt, float) - np.asarray(ph, float) + 0.5, 1.0) - 0.5
    m = p.K["PH_SLEW"]
    return np.mod(np.asarray(ph, float) + np.clip(d, -m, m), 1.0)


def kin_step_vec(cmd, T_cycle: float, p: Params) -> np.ndarray:
    ref, XY = p.ref, p.T["FOOT_XY_BODY"]
    rot = ref["rot_step_frac"] * cmd[2]
    dx = T_cycle * (-rot * XY[:, 1])
    dy = T_cycle * (ref["lat_cmd_gain"] * cmd[1] + rot * XY[:, 0])
    return np.stack([dx, dy], 1)


def posture_offset(cmd, A: dict, p: Params) -> np.ndarray:
    ref, T = p.ref, p.T
    pa = ref["posture_arc"]
    inner = np.array([0.0, 1.0, 0.0, 1.0]) if cmd[2] > 0 else np.array([1.0, 0.0, 1.0, 0.0])
    amount = T["FRONT"] * pa["front"] + (1.0 - T["FRONT"]) * pa["rear"]
    arc = A["arc"] * inner * amount * (-T["SIDE_Y"])
    lat = A["lat"] * ref.get("posture_lat", 0.0) * T["SIDE_Y"]
    return arc + lat


def wheel_cmd(cmd, A: dict, res, p: Params) -> np.ndarray:
    ref, T = p.ref, p.T
    diff = (cmd[0] - cmd[2] * T["SIDE_Y"] * ref["L_eff"] / 2) / ref["r_wheel"]
    dom = _mirror(T["DOM_TURN_L"], cmd[2] < 0, p)
    gate = 1.0 + A["turn"] * (1.0 - A["wheel"]) * (1.0 - 2.0 * dom)
    lw = _mirror(T["LAT_WHEEL_L"], cmd[1] < 0, p) * ref["lat_omega"] * float(np.clip(abs(cmd[1]) / ref["lat_vy_ref"], 0.0, 1.5))
    return diff * gate + lw + np.asarray(res, float)


def step_pattern(cmd, p: Params) -> dict:
    ref = p.ref
    cmd = np.asarray(cmd, float)
    A = activity(cmd, p)
    s = leg_activity(cmd, A, p)
    wl = lat_weight(A)
    cyc_on = wl * float(ref["step_gen_lat"] == "cycle") + (1.0 - wl) * float(ref["step_gen_turn"] == "cycle")
    A_kin = dict(A, s4=A["s4"] * (1.0 - cyc_on))
    s_kin = leg_activity(cmd, A_kin, p)
    hz = wl * ref["step_hz_lat"] + (1.0 - wl) * ref["step_hz_turn"]
    duty = wl * ref["duty_lat"] + (1.0 - wl) * ref["duty_turn"]
    lift = wl * ref["lift_lat"] + (1.0 - wl) * ref["lift"]
    return dict(A=A, s=s, s_kin=s_kin, g=step_gain(s_kin, p), vec=kin_step_vec(cmd, 1.0 / hz, p), ph=phase_offsets(cmd, A, p),
                wheel0=wheel_cmd(cmd, A, np.zeros(4), p), hz=hz, duty=duty, cyc_on=cyc_on,
                lift=lift * (s_kin / max(float(np.max(s_kin)), 1e-6)) * step_gain(s_kin, p), post_y=posture_offset(cmd, A, p))


def duty_remap(th, duty: float) -> np.ndarray:
    ph = np.mod(np.asarray(th, float), TWO_PI) / TWO_PI
    sw = 1.0 - duty
    return np.where(ph < sw, np.pi * ph / sw, np.pi + np.pi * (ph - sw) / duty)


def act_split(a, p: Params) -> dict:
    K = p.K
    a = np.tanh(np.asarray(a, float))
    return dict(om=1.0 + K["OMEGA_NOM_SCALE"] * a[0], amp=1.0 + K["AMP_SCALE"] * a[1:5], lift=1.0 + K["LIFT_SCALE"] * a[5],
                sway=K["SWAY_MAX"] * a[6:8], wres=K["WHEEL_RES"] * a[8:12], qres=K["QRES_MAX"] * a[12:24],
                foot_x=K["FOOT_OFF_MAX"] * a[1:5], foot_z=K["FOOT_OFF_MAX"] * a[5])


def foot_targets(theta, amp, g, vec, lift4, sway, u, foot_x, foot_z, duty, post_y, z_sag, p: Params) -> np.ndarray:
    th = duty_remap(theta, duty)
    s = np.sin(th)
    prog = -0.5 * np.cos(th)
    amp = np.asarray(amp, float); g = np.asarray(g, float)
    ramp = np.clip(amp, 0.0, 1.0)
    dxy = (np.asarray(vec, float) * (amp * g)[:, None]) * prog[:, None]
    dz = np.where(s > 0, (np.asarray(lift4, float) + z_sag * g) * s, 0.0) * ramp
    step_off = np.concatenate([dxy, dz[:, None]], 1) + u * np.array([sway[0], sway[1], 0.0])
    post = np.stack([np.zeros(4), np.asarray(post_y, float), np.zeros(4)], 1)
    wheel_off = np.stack([np.asarray(foot_x, float), np.zeros(4), np.full(4, float(foot_z))], 1)
    return p.T["STANCE_FEET"] + post + step_off + (1.0 - u) * wheel_off


def joint_targets(feet, p: Params) -> np.ndarray:
    ks = p.T["KNEE_SIGN4"]
    out = np.zeros(12)
    for k in range(4):
        (q1, q2, q3), _ = kin.ik(LEG_SHM[k], float(feet[k, 0]), float(feet[k, 1]), float(feet[k, 2]), float(ks[k]))
        out[3 * k:3 * k + 3] = (q1, q2, q3)
    return out


def cyc_interp(table, phi: float, p: Params) -> np.ndarray:
    n = p.K["CYC_N"]
    x = np.mod(phi / TWO_PI, 1.0) * n
    i0 = int(np.floor(x)) % n
    i1 = (i0 + 1) % n
    f = x - np.floor(x)
    return table[:, i0] * (1.0 - f) + table[:, i1] * f


def cycle_offsets(phi: float, cmd, A: dict, p: Params) -> dict:
    ref, T = p.ref, p.T
    off = cyc_interp(T["CYC_OFF"], phi, p)
    tau = cyc_interp(T["CYC_TAU"], phi, p)
    w_ll = 0.5 * (1.0 + float(np.clip(cmd[1] / 0.02, -1.0, 1.0)))
    w_tl = 0.5 * (1.0 + float(np.clip(cmd[2] / 0.3, -1.0, 1.0)))
    lo, hi = ref["cyc_lat_amp_clip"]
    amp_l = float(np.clip(ref["cyc_lat_amp0"] + ref["cyc_lat_slope"] * (abs(cmd[1]) - ref["cyc_lat_vy0"]), lo, hi)) * ref["cyc_amp_lat"]
    amp_t = float(np.clip(abs(cmd[2]) / ref["a_ref"]["wz"], 0.3, 1.2)) * ref["cyc_amp_turn"]
    on_l, on_t = float(ref["step_gen_lat"] == "cycle"), float(ref["step_gen_turn"] == "cycle")
    wl = lat_weight(A)

    def blend(Tb, sc_l=1.0, sc_t=1.0):
        lat = w_ll * Tb[0] + (1.0 - w_ll) * Tb[1]
        turn = w_tl * Tb[2] + (1.0 - w_tl) * Tb[3]
        return wl * amp_l * on_l * lat * sc_l + (1.0 - wl) * amp_t * on_t * turn * sc_t

    sc_lat = np.tile(np.asarray(ref["cyc_joint_scale_lat"], float), 4)
    sc_turn = np.tile(np.asarray(ref["cyc_joint_scale_turn"], float), 4)
    delta = blend(off, sc_lat, sc_turn)
    wheel_tau = blend(tau) * ref["cyc_wheel"]
    HZ = T["CYC_HZ"]
    hz = wl * (w_ll * HZ[0] + (1.0 - w_ll) * HZ[1]) + (1.0 - wl) * (w_tl * HZ[2] + (1.0 - w_tl) * HZ[3])
    return dict(delta=delta, wheel_tau=wheel_tau, hz=hz * ref["cyc_hz_scale"], on=wl * on_l + (1.0 - wl) * on_t)


def wheel_dead(tau, p: Params) -> np.ndarray:
    """τ 死區（= rl_env_v3.wheel_ctrl_tau 的力矩部分）：|τ| < TAU_FF → 0，否則 τ + TAU_FF·sign。寫進 effort 欄前才套。"""
    tau = np.asarray(tau, float)
    ff = p.K["TAU_FF"]
    return np.where(np.abs(tau) < ff, 0.0, tau + np.sign(tau) * ff)

def _env_rollout_ctrl(env, cmd, steps, acts, seed=0):
    """env 跑 steps 步（固定 cmd、給定動作序列），回每步的 (ctrl_leg12, ctrl_wheel4, v_wheel_meas4, delay, c)。
    q_des 就是 pipeline_state.ctrl[LEG_ACT_IDX]；輪 ctrl = v_meas + τ_dead/kv（wheel_ctrl_tau）。"""
    jr, js = jax.jit(env.reset), jax.jit(env.step)
    s = jr(jax.random.PRNGKey(seed))
    s = s.replace(info={**s.info, "cmd": jnp.array(cmd, jnp.float32), "cmd2": jnp.array(cmd, jnp.float32), "t_switch": 10 ** 6})
    delay = int(s.info["delay"])
    out = []
    for i in range(steps):
        v_meas = np.asarray(s.pipeline_state.qvel[v3.WHEEL_QVEL_IDX])
        s = js(s, jnp.array(acts[i], jnp.float32))
        d = s.pipeline_state
        out.append(dict(q_des=np.asarray(d.ctrl[v3.LEG_ACT_IDX]), wctrl=np.asarray(d.ctrl[v3.WHEEL_ACT_IDX]), v_meas=v_meas,
                        c={k: np.asarray(v) for k, v in s.info["c"].items()}, s4=float(s.info["s4"]), s_arc=float(s.info["s_arc"])))
    return out, delay

# ---------------------------------------------------------------- 產生器狀態機（對應 DualModeEnv.reset／step 的產生器部分）
class Gen:
    """v3 產生器。狀態 = env `info` 裡產生器會用到的那幾個；不含延遲、EMA、reward。"""

    def __init__(self, p: Params, cmd0=(0.0, 0.0, 0.0)):
        self.p = p
        K = p.K
        self.dt = K["CTRL_DT"]
        cmd0 = np.asarray(cmd0, float)
        P0 = step_pattern(cmd0, p)
        self.c = dict(theta=np.zeros(4), amp=np.zeros(4), wheel=np.zeros(4), cyc_abad=np.zeros(4), cyc_knee=np.zeros(4))
        self.u_mode = float(np.max(P0["s"]))
        self.ph = P0["ph"].copy()
        self.s4, self.s_arc = P0["A"]["s4"], P0["A"]["arc"]
        self.phi_cyc = 0.0
        self.u4 = 0.0
        self.qres = np.zeros(12)
        self.sway = np.zeros(2)
        self.head_err = 0.0
        self.cmd_prev = cmd0
        self.i = 0

    def stance_q12(self) -> np.ndarray:
        return self.p.T["STANCE_Q12"].copy()

    def integrate_head(self, gyro_z: float) -> None:
        if self.i > 0:
            self.head_err += (float(gyro_z) - float(self.cmd_prev[2])) * self.dt

    def step(self, cmd, act, v_wheel):
        p, K, ref, T = self.p, self.p.K, self.p.ref, self.p.T
        cmd = np.asarray(cmd, float); act = np.asarray(act, float); v_wheel = np.asarray(v_wheel, float)
        P = step_pattern(cmd, p)
        u_tgt = float(np.max(P["s"]))
        du = self.dt / ref["trans_s"]
        u_mode = float(np.clip(self.u_mode + np.clip(u_tgt - self.u_mode, -du, du), 0.0, 1.0))
        u_ramp = min(1.0, self.i / K["RAMP_STEPS"])
        A = act_split(u_ramp * act, p)
        om = P["hz"] * A["om"]
        ph = slew_phase(self.ph, P["ph"], p)
        theta_free = self.c["theta"] + TWO_PI * om * self.dt
        ph_rad = TWO_PI * ph
        phi0 = theta_free[0] - ph_rad[0]
        theta = np.mod(phi0 + ph_rad if u_mode > 0.01 else self.c["theta"], TWO_PI)
        amp = self.c["amp"] + np.clip(P["g"] * A["amp"] - self.c["amp"], -0.05, 0.05)
        sway = self.sway + np.clip(A["sway"] * u_mode - self.sway, -K["SWAY_SLEW"], K["SWAY_SLEW"])
        lift4 = P["lift"] * A["lift"]
        feet = foot_targets(theta, amp, P["g"], P["vec"], lift4, sway, u_mode, A["foot_x"], A["foot_z"], P["duty"], P["post_y"], ref["z_sag"], p)
        u4_tgt = float(np.clip(P["A"]["s4"] / 0.3, 0.0, 1.0)) * P["cyc_on"]
        u4 = float(np.clip(self.u4 + np.clip(u4_tgt - self.u4, -du, du), 0.0, 1.0))
        CY = cycle_offsets(self.phi_cyc, cmd, P["A"], p)
        phi_cyc = np.mod(self.phi_cyc + TWO_PI * CY["hz"] * A["om"] * self.dt * float(u4 > 0.01), TWO_PI)
        delta = (CY["delta"].reshape(4, 3) * A["amp"][:, None] * np.array([1.0, A["lift"], A["lift"]])[None, :]).reshape(12) * u4
        qres = self.qres + np.clip(A["qres"] - self.qres, -K["QRES_SLEW"], K["QRES_SLEW"])
        q_des = np.clip(joint_targets(feet, p) + delta + qres, T["LO12"], T["HI12"])
        v_nom = wheel_cmd(cmd, P["A"], np.zeros(4), p)
        tau = ref["wheel_outer_gain"] * (v_nom - v_wheel) + A["wres"] + u4 * CY["wheel_tau"]
        d3 = delta.reshape(4, 3)
        self.c = dict(theta=theta, amp=amp, wheel=tau, cyc_abad=d3[:, 0].copy(), cyc_knee=d3[:, 2].copy())
        self.u_mode, self.ph, self.u4, self.phi_cyc, self.qres, self.sway = u_mode, ph, u4, phi_cyc, qres, sway
        self.s4, self.s_arc = P["A"]["s4"], P["A"]["arc"]
        self.cmd_prev = cmd
        self.i += 1
        return q_des, tau, self.c
```

# task7/realbot/rl_obs_v3.py
"""狗上純 numpy 的 v3 observation（88 維）組裝 ＋ 左右鏡像（D1 Max，2026-09-23）。

只 import numpy 與 realbot/ 內檔案。與 `rl_env_v3.DualModeEnv._obs` 的對應由 `tests/test_rl_obs_v3.py` 釘住。

obs 88：grav 3 | gyro 3 | q−stance 12 | qvel 12 | wheel_v 4 | cmd 3 | [s4, s_arc] 2 | head_err 1 | last_a 24 |
        amp 4 | sinθ 4 | cosθ 4 | cyc_abad 4 | cyc_knee 4 | wheel_τ/5 4
  - 關節：馬達座標 → 控制器座標 `coord.to_ctrl`，**減站姿 stance_q12（不是 HOME12）**；速度除 SIGN；MJCF 腿序
  - 輪速：`joint_state.velocity` × `coord.SIGN[KIND_WHEEL]`（馬達 → MJCF 方向）；MJCF 序 (FR, FL, RR, RL)
  - 四元數 xyzw → wxyz → `w2b(q, (0,0,−1))`；gyro 直接用（IMU 偏置不修正，訓練有隨機化）
  - 站姿與鏡像索引表由 npz 帶，本檔零常數（腿名與輪名除外）
鏡像（`mirror_obs`／`mirror_act`）：右向指令時把 obs 左右翻給 policy、動作翻回來 —— 等於 `rl_env_v3.mirror_policy(turn_only=False)`。
"""
from __future__ import annotations

import numpy as np

import coord
import rl_obs
from rl_obs import Frame, w2b  # noqa: F401  （M9 沿用同一個 Frame）

LEG_NAMES = rl_obs.LEG_NAMES                                              # 12，MJCF 序
WHEEL_NAMES = [rl_obs.LEG_SHM[l] + coord.KIND_WHEEL for l in rl_obs.LEG_MJCF]   # 4，MJCF 序
WHEEL_SIGN4 = np.array([coord.SIGN[coord.KIND_WHEEL][n[:2]] for n in WHEEL_NAMES])
ACT_DIM, OBS_DIM = 24, 88
_DOWN = np.array([0.0, 0.0, -1.0])
_SIGN12 = rl_obs._SIGN12


def read_frame(state_ro, imu_shm, idx: dict, state_stride: int) -> Frame:
    """狗上用：12 腿關節 ＋ 4 輪 ＋ imu_central → Frame。`idx` 要含 16 個關節名。"""
    st = state_ro.states()
    names = LEG_NAMES + WHEEL_NAMES
    pos = {n: st[idx[n]]["position"] for n in names}
    vel = {n: st[idx[n]]["velocity"] for n in names}
    imu = imu_shm.imu()
    return Frame(pos, vel, imu["quat"], imu["gyro"], state_ro.read_tick(state_stride))


def is_right(cmd) -> bool:
    """鏡像觸發：|wz| > 0.1 時看 wz < 0，否則看 vy < −0.02（= mirror_policy(turn_only=False)）。"""
    vx, vy, wz = (float(x) for x in cmd)
    return (wz < 0.0) if abs(wz) > 0.1 else (vy < -0.02)


class Obs:
    def __init__(self, npz_path: str):
        with np.load(npz_path, allow_pickle=False) as z:
            self.stance12 = np.asarray(z["STANCE_Q12"], dtype=float)
            self.oi = np.asarray(z["OBS_MIRROR_IDX"]).astype(int)
            self.os = np.asarray(z["OBS_MIRROR_SGN"], dtype=np.float32)
            self.ai = np.asarray(z["ACT_MIRROR_IDX"]).astype(int)
            self.as_ = np.asarray(z["ACT_MIRROR_SGN"], dtype=float)
            self.lr = np.asarray(z["MIRROR_LR"]).astype(int)
        assert self.stance12.shape == (12,) and self.oi.shape == (OBS_DIM,) and self.ai.shape == (ACT_DIM,)

    def wheel_v(self, fr: Frame) -> np.ndarray:
        return np.array([fr.vel[n] for n in WHEEL_NAMES], dtype=float) * WHEEL_SIGN4

    @staticmethod
    def gyro_z(fr: Frame) -> float:
        return float(fr.gyro[2])

    def build(self, fr: Frame, c: dict, cmd, last_a, head_err: float, s4: float, s_arc: float) -> np.ndarray:
        cmd = np.asarray(cmd, dtype=np.float64).reshape(-1)
        last_a = np.asarray(last_a, dtype=np.float64).reshape(-1)
        assert cmd.size == 3, f"cmd 應為 3 維 (vx, vy, wz)，實得 {cmd.size}"
        assert last_a.size == ACT_DIM, f"last_a 應為 {ACT_DIM} 維，實得 {last_a.size}"
        q = np.asarray(fr.quat_xyzw, dtype=float)[[3, 0, 1, 2]]
        jpos = np.array([coord.to_ctrl(n, fr.pos[n]) for n in LEG_NAMES], dtype=float)
        jvel = np.array([fr.vel[n] for n in LEG_NAMES], dtype=float) / _SIGN12
        th = np.asarray(c["theta"], dtype=float)
        o = np.concatenate([
            w2b(q, _DOWN), np.asarray(fr.gyro, dtype=float),
            jpos - self.stance12, jvel, self.wheel_v(fr),
            cmd, [float(s4), float(s_arc)], [float(np.clip(head_err, -1.0, 1.0))],
            last_a,
            np.asarray(c["amp"], float), np.sin(th), np.cos(th), np.asarray(c["cyc_abad"], float), np.asarray(c["cyc_knee"], float),
            np.asarray(c["wheel"], float) / 5.0,
        ]).astype(np.float32)
        assert o.size == OBS_DIM, f"obs 應為 {OBS_DIM} 維，實得 {o.size}"
        return o

    def mirror_obs(self, obs) -> np.ndarray:
        """照 rl_env_v3.mirror_obs：索引／符號表 ＋ θ 相位修正（腿交換後四腿一起加 d = θ0 − θ1）。"""
        obs = np.asarray(obs, dtype=np.float32)
        o = obs[self.oi] * self.os
        th = np.arctan2(obs[68:72], obs[72:76])
        d = th[0] - th[1]
        th_m = th[self.lr] + d
        o[68:72] = np.sin(th_m)
        o[72:76] = np.cos(th_m)
        return o.astype(np.float32)

    def mirror_act(self, a) -> np.ndarray:
        a = np.asarray(a, dtype=float)
        return a[self.ai] * self.as_

    """輪阻尼按階段：步態段用 --wheel-kd；其餘段 v3 模式一律 WHEEL_KD_SAFE（0.5，起身承重滾動要阻尼），

def v3_wheel_rows(tau_mjcf4, v_meas_motor4, wheel_kd: float, p) -> list:

    return [dict(velocity=float(v_meas_motor4[k]), effort=float(tau[k]), kp=0.0, kd=float(wheel_kd)) for k in range(4)]

```python
class V3GaitStream(GaitStream):
    """CPG-RL v3.6f 步態流（2026-09-23）：每個 50 Hz 推進點
    讀 shm（12 腿＋4 輪＋IMU）→ `rl_obs_v3.Obs.build` → （右向指令）鏡像 → `policy_np` → 鏡像回 → `gen_v3.Gen.step`
    → 12 關節目標（線性內插到 200 Hz）＋ 4 輪 τ（零階保持，主迴圈寫 effort）。
    與本機 jax 堆疊的對應由 `tests/test_policy_v3_closed_loop.py` 影子跟跑釘住。

    ★ 退路是**零動作名目**（不是 A 步態）：當步異常（obs NaN／幀沒前進／推論超時／例外）→ 該步動作 0；
      連續 FALLBACK_LIMIT 步 → 永久零動作；鍵 `o` → 永久零動作＋cmd 0。gain 0 ＝ 乾跑（obs、推論照跑照記，動作 0）。
    指令時序：進 GAIT_IN 後前 CMD_DELAY_STEPS（1 s）cmd=0（產生器輸出站姿，與 smoothstep 淡入相容），之後切到 --cmd；
    `set_open_loop`（Enter 停走進 GAIT_OUT 時 InteractivePlan 會呼叫）後 cmd=0，產生器自己在 trans_s 內收回站姿。
    **不加動作延遲、不加雜訊** —— 實機的是真的。
    """

    FALLBACK_LIMIT = 25
    CMD_DELAY_STEPS = 50
    INFER_BUDGET_MS = 10.0

    def __init__(self, gen, obs_builder, policy, reader, cmd3, gain: float, mirror: bool, log: list = None):
        import numpy as np
        self._np = np
        self.gen, self.O, self.policy, self.reader = gen, obs_builder, policy, reader
        self.cmd_target = np.asarray(cmd3, dtype=float).reshape(3)
        self.gain, self.mirror = float(gain), bool(mirror)
        self.last_a = np.zeros(24)
        self.wheel_tau_mjcf = np.zeros(4)
        self.i = 0
        self.n_fallback = self.n_fallback_total = 0
        self.open_loop, self.open_loop_why = False, ""
        self.last_tick = None
        self.log = log if log is not None else []
        self.worst_ms = 0.0
        self.seg = 1
        self.n_clamp = 0
        self.t_next = 0.0
        self._q12 = gen.stance_q12()
        self.q_prev = self.q_next = self._compute()

    def _compute(self) -> dict:
        import rl_obs
        return {n: float(self._q12[k]) for k, n in enumerate(rl_obs.LEG_NAMES)}

    def set_open_loop(self, why: str) -> None:
        if not self.open_loop:
            self.open_loop, self.open_loop_why = True, why
            print(f"\n  ▸ policy 退回零動作名目、指令歸零（{why}）—— 產生器繼續\n")

    def _cmd_now(self):
        if self.open_loop or self.i < self.CMD_DELAY_STEPS:
            return self._np.zeros(3)
        return self.cmd_target

    def _advance(self) -> None:
        import time
        import rl_obs_v3
        np = self._np
        t0 = time.perf_counter()
        fb, obs, a, imu, right, fr = "", None, None, None, False, None
        cmd = self._cmd_now()
        try:
            fr = self.reader()
            imu = (list(map(float, fr.quat_xyzw)), list(map(float, fr.gyro)))
            if self.last_tick is not None and fr.tick == self.last_tick:
                fb = "stale"
            self.last_tick = fr.tick
            self.gen.integrate_head(self.O.gyro_z(fr))
            obs = self.O.build(fr, self.gen.c, self.gen.cmd_prev, self.last_a, self.gen.head_err, self.gen.s4, self.gen.s_arc)
            if not np.all(np.isfinite(obs)):
                fb = fb or "obs_nan"
            right = self.mirror and rl_obs_v3.is_right(cmd)
            a = self.O.mirror_act(self.policy.infer(self.O.mirror_obs(obs))) if right else self.policy.infer(obs)
            if not np.all(np.isfinite(a)):
                fb = fb or "act_nan"
        except Exception as e:            # noqa: BLE001 —— 承重中不能讓例外炸掉主迴圈
            fb = fb or f"exc:{type(e).__name__}"
        ms = (time.perf_counter() - t0) * 1000.0
        if not fb and ms > self.INFER_BUDGET_MS:
            fb = f"slow:{ms:.1f}ms"
        if fb:
            self.n_fallback += 1
            self.n_fallback_total += 1
            if self.n_fallback >= self.FALLBACK_LIMIT:
                self.set_open_loop(f"連續 {self.n_fallback} 步退回（最後：{fb}）")
        else:
            self.n_fallback = 0
        a_use = np.zeros(24) if (fb or a is None or self.open_loop or self.gain == 0.0) else self.gain * np.asarray(a, float)
        v_wheel = self.O.wheel_v(fr) if fr is not None else np.zeros(4)
        if fr is not None and not np.all(np.isfinite(v_wheel)):
            v_wheel = np.zeros(4)
        q12, tau, _ = self.gen.step(cmd, a_use, v_wheel)
        ms = (time.perf_counter() - t0) * 1000.0
        self.worst_ms = max(self.worst_ms, ms)
        self._q12 = q12
        self.wheel_tau_mjcf = tau
        self.last_a = a_use
        self.log.append({"i": self.i, "seg": self.seg, "tg": round(self.t_next, 3),
                         "cmd": [float(x) for x in cmd], "right": bool(right),
                         "obs": None if obs is None else [round(float(x), 5) for x in obs],
                         "a": [round(float(x), 5) for x in a_use],
                         "q_des": [round(float(x), 5) for x in q12], "tau": [round(float(x), 4) for x in tau],
                         "v_wheel": [round(float(x), 3) for x in v_wheel], "head": round(float(self.gen.head_err), 5),
                         "imu": imu, "ms": round(ms, 3), "fb": fb, "ol": self.open_loop})
        self.i += 1
```
