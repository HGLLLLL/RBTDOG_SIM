```python
"""釘住 v1b 的 WheelEnv() 預設路徑（obs／reward／qpos／ctrl 逐位元）。

"""
import sys
from pathlib import Path

import numpy as np
import pytest

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "inference"))
jax = pytest.importorskip("jax")
jnp = jax.numpy
import rl_env_wheel as rw  # noqa: E402

GOLDEN = Path(__file__).resolve().parent / "data" / "rl_env_wheel_golden.npz"
SEEDS = (0, 1, 2)      # 三個 reset：出生方塊、指令、雜訊都不同
STEPS = 12


def _rollout():
    """WheelEnv() 全預設（地形、雜訊、推力、隨機指令），固定 seed、固定動作序列。"""
    env = rw.WheelEnv()
    jit_reset, jit_step = jax.jit(env.reset), jax.jit(env.step)
    O, P, R, Q, C = [], [], [], [], []
    for sd in SEEDS:
        s = jit_reset(jax.random.PRNGKey(sd))
        o, p, r, q, c = [], [], [], [], []
        for i in range(STEPS):
            a = jnp.sin(jnp.arange(rw.ACT_DIM) * 0.7 + i * 0.3 + sd) * 0.5     # 非零、可重現、四腿都動
            s = jit_step(s, a)
            o.append(np.asarray(s.obs["state"])); p.append(np.asarray(s.obs["privileged_state"])); r.append(float(s.reward))
            q.append(np.asarray(s.pipeline_state.qpos)); c.append(np.asarray(s.pipeline_state.ctrl))
        O.append(o); P.append(p); R.append(r); Q.append(q); C.append(c)
    return np.array(O), np.array(P), np.array(R), np.array(Q), np.array(C)


def test_v1b_default_rollout_is_bitwise_unchanged():
    obs, priv, rew, qpos, ctrl = _rollout()
    if not GOLDEN.exists():
        GOLDEN.parent.mkdir(parents=True, exist_ok=True)
        np.savez(GOLDEN, obs=obs, priv=priv, reward=rew, qpos=qpos, ctrl=ctrl)
        pytest.skip(f"golden 不存在，已產生 {GOLDEN} —— 請 commit 後重跑")
    g = np.load(GOLDEN)
    for name, got in (("obs", obs), ("priv", priv), ("reward", rew), ("qpos", qpos), ("ctrl", ctrl)):
        np.testing.assert_allclose(got, g[name], rtol=0, atol=0, err_msg=f"{name} 與 golden 不同 —— v1b 預設路徑被動到了")


def test_golden_actually_moves():
    """三個 seed 的輪 ctrl 都非零、obs 有雜訊差異，否則擋不住回歸。"""
    if not GOLDEN.exists():
        pytest.skip("golden 尚未產生")
    g = np.load(GOLDEN)
    assert np.abs(g["ctrl"][:, :, -4:]).max(axis=(1, 2)).min() > 0.1
    assert np.abs(g["obs"][:, :, :53] - g["priv"][:, :, :53]).max() > 1e-3     # state 帶雜訊、priv 乾淨
```

```python
import sys
from pathlib import Path

import numpy as np
import pytest

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "inference"))
jax = pytest.importorskip("jax")

from mujoco import mjx  # noqa: E402


@pytest.fixture(scope="module")
def env():
    return rw.WheelEnv(flat=True, noise=False, push=False, fixed_cmd=True)


def _walked_state(env, n=6):
    """從 reset 走幾步非對稱動作，得到一個左右不對稱、四輪觸地的狀態。"""
    s = jax.jit(env.reset)(jax.random.PRNGKey(0))
    step = jax.jit(env.step)
    for i in range(n):
        a = jnp.sin(jnp.arange(16) * 0.9 + i * 0.4) * 0.3
        s = step(s, a)
    return s


def test_mirror_maps_are_involutions():
    o = jnp.arange(53, dtype=jnp.float32) + 1.0
    a = jnp.arange(16, dtype=jnp.float32) + 1.0
    p = jnp.arange(56, dtype=jnp.float32) + 1.0
    assert np.allclose(rw.mirror_obs(rw.mirror_obs(o)), o) and np.allclose(rw.mirror_act(rw.mirror_act(a)), a)
    assert np.allclose(rw.mirror_priv(rw.mirror_priv(p)), p)
    assert not np.allclose(rw.mirror_obs(o), o)
    # 符號表：ωx ωz、grav_y、cmd vy wz、ABAD（jpos／jvel／last_a／act）變號；其餘 +1
    sg = np.asarray(rw.OBS_MIRROR_SGN)
    assert sg[0] == -1 and sg[1] == 1 and sg[2] == -1 and sg[4] == -1 and sg[7] == -1 and sg[8] == -1
    assert np.all(sg[9:13] == -1) and np.all(sg[13:21] == 1) and np.all(sg[21:25] == -1) and np.all(sg[25:37] == 1)
    assert np.all(sg[37:41] == -1) and np.all(sg[41:53] == 1)
    assert np.all(np.asarray(rw.ACT_MIRROR_SGN)[:4] == -1) and np.all(np.asarray(rw.ACT_MIRROR_SGN)[4:] == 1)
    idx = np.asarray(rw.OBS_MIRROR_IDX)
    assert list(idx[9:13]) == [10, 9, 12, 11] and list(idx[33:37]) == [34, 33, 36, 35]     # 腿 FR↔FL、RR↔RL（輪速也換）


def test_mirror_obs_matches_physically_mirrored_state(env):
    s = _walked_state(env)
    d = s.pipeline_state
    cmd, la = jnp.array([0.3, 0.2, -0.4]), jnp.sin(jnp.arange(16)) * 0.5
    o = env._obs(d, cmd, la)
    dm = rw.mirror_data(env, d)
    om = env._obs(dm, cmd * jnp.array([1.0, -1.0, -1.0]), rw.mirror_act(la))
    np.testing.assert_allclose(np.asarray(om), np.asarray(rw.mirror_obs(o)), atol=1e-5)
    vb = rw.quat_rot_inv(d.qpos[3:7], d.qvel[0:3]); vbm = rw.quat_rot_inv(dm.qpos[3:7], dm.qvel[0:3])
    np.testing.assert_allclose(np.asarray(vbm), np.asarray(vb) * np.array([1, -1, 1]), atol=1e-5)


def test_physics_is_left_right_equivariant(env):
    """翻過的狀態用翻過的動作跑 5 步 ＝ 原軌跡翻過去（同時驗證模型左右對稱）。"""
    s = _walked_state(env)
    d, dm = s.pipeline_state, rw.mirror_data(env, s.pipeline_state)
    phys = jax.jit(env._physics)
    for i in range(5):
        a = jnp.sin(jnp.arange(16) * 0.5 + i) * 1.5
        d = phys(d, a); dm = phys(dm, rw.mirror_act(a))
    dmm = rw.mirror_data(env, dm)
    np.testing.assert_allclose(np.asarray(dmm.qpos), np.asarray(d.qpos), atol=1e-4)
    np.testing.assert_allclose(np.asarray(dmm.qvel), np.asarray(d.qvel), atol=1e-3)
```

```python
# ---------------------------------------------------------------- 左右鏡像（v1c 鏡像增強；y → −y）
# 腿 FR↔FL、RR↔RL。四腿關節軸同向（ABAD x 軸、髖／膝／輪 y 軸），y 鏡射下繞 x／z 的量變號、繞 y 的不變：
# ABAD 位置／速度／動作、ωx／ωz、grav_y、cmd vy／wz、機身 vy 變號；髖、膝、輪速、輪動作只換腿。
MIRROR_LR = (1, 0, 3, 2)


def _mirror_maps():
    oi, os_ = np.arange(OBS_DIM), np.ones(OBS_DIM, np.float32)
    ai, as_ = np.arange(ACT_DIM), np.ones(ACT_DIM, np.float32)

    def group4(idx, sg, base, neg):
        for k in range(4):
            idx[base + k] = base + MIRROR_LR[k]
            if neg:
                sg[base + k] = -1.0
    os_[0] = os_[2] = -1.0                      # ωx、ωz
    os_[4] = -1.0                               # grav y
    os_[7] = os_[8] = -1.0                      # cmd vy、wz
    for base, neg in ((9, True), (13, False), (17, False)):                  # jpos：ABAD／HIP／KNEE
        group4(oi, os_, base, neg)
    for base, neg in ((21, True), (25, False), (29, False), (33, False)):    # jvel：＋WHEEL
        group4(oi, os_, base, neg)
    for base, neg in ((37, True), (41, False), (45, False), (49, False)):    # last_action
        group4(oi, os_, base, neg)
    for base, neg in ((0, True), (4, False), (8, False), (12, False)):       # 動作
        group4(ai, as_, base, neg)
    return jnp.array(oi), jnp.array(os_), jnp.array(ai), jnp.array(as_)


OBS_MIRROR_IDX, OBS_MIRROR_SGN, ACT_MIRROR_IDX, ACT_MIRROR_SGN = _mirror_maps()
PRIV_MIRROR_SGN = jnp.array([1.0, -1.0, 1.0], dtype=jnp.float32)      # 機身系線速度 (vx, vy, vz)


def mirror_obs(o):
    return o[OBS_MIRROR_IDX] * OBS_MIRROR_SGN


def mirror_act(a):
    return a[ACT_MIRROR_IDX] * ACT_MIRROR_SGN


def mirror_priv(p):
    return jnp.concatenate([mirror_obs(p[:OBS_DIM]), p[OBS_DIM:] * PRIV_MIRROR_SGN])


def mirror_data(env, data):
    """測試用：把 MJX data 真的左右翻（y → −y、四元數 (w,x,y,z) → (w,−x,y,−z)、腿互換、ABAD 變號、ctrl 同樣翻），再 mjx.forward。
    free joint 的 qvel 線速度在世界系（vy 變號）、角速度在機身系（ωx、ωz 變號）。"""
    q, v = data.qpos, data.qvel
    q = q.at[1].multiply(-1.0).at[4].multiply(-1.0).at[6].multiply(-1.0)
    v = v.at[1].multiply(-1.0).at[3].multiply(-1.0).at[5].multiply(-1.0)
    QA, VA = env.QADR, env.VADR
    jq, jv = data.qpos[QA], data.qvel[VA]
    sgn16 = jnp.concatenate([-jnp.ones(4), jnp.ones(12)])
    idx16 = jnp.array([g * 4 + MIRROR_LR[k] for g in range(4) for k in range(4)])
    q = q.at[QA].set(jq[idx16] * sgn16); v = v.at[VA].set(jv[idx16] * sgn16)
    c = data.ctrl.at[env.AID].set(data.ctrl[env.AID][idx16] * sgn16)        # PD 目標／輪速也翻（ABAD offset 0，變號合法）
    return mjx.forward(env.sys, data.replace(qpos=q, qvel=v, ctrl=c))
```

```python
def bias_matrix(roll, pitch):
    """IMU 常數偏差 R ＝ Ry(pitch)·Rx(roll)（與 realbot/rl_obs_rl.grav_fix_matrix 同慣例）；policy 看到 g ＝ Rᵀ·g_true，狗端 --grav-offset 乘回 R 就歸零。"""
```

```python
def landing_vz(clr, dt, thr=0.005):
    """落地衝擊（v1c 及格線用）：每腿「淨空由 > thr 降到 ≤ thr」算一次觸地，取觸地前一步的下落速度 (clr[t−2] − clr[t−1]) / dt（m/s，負值截 0）。
    clr：(T, 4) m。回傳 (事件平均 vz, 事件數)；與 WheelEnv 的 impact 項同口徑（觸地前一步的 vz）。"""
    ev = []
    for k in range(clr.shape[1]):
        c = clr[:, k]
        for t in range(2, len(c)):
            if c[t] <= thr and c[t - 1] > thr:
                ev.append(max(0.0, float(c[t - 2] - c[t - 1]) / dt))
    return (float(np.mean(ev)) if ev else 0.0), len(ev)
```
